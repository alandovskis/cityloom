import { appendFileSync, writeFileSync } from "node:fs";

const OUT = "/private/tmp/claude-501/-Users-alex-src-cityloom/4cb51d1a-df55-4d28-90b1-a94549351273/scratchpad/ev63";
const mk = (tag: string) => {
  const f = `${OUT}/log-${tag}.txt`;
  writeFileSync(f, "");
  return (s: string) => appendFileSync(f, s + "\n");
};
const norm = (s: string) => s.replace(/\s+/g, " ").trim();

async function full(page: Page, sel: string, file: string, log: (s: string) => void, nth = 0) {
  const vp = page.viewportSize()!;
  await page.setViewportSize({ width: vp.width, height: 3600 });
  await page.waitForTimeout(400);
  const loc = page.locator(sel).nth(nth);
  await loc.scrollIntoViewIfNeeded();
  const m = await loc.evaluate((e) => ({ client: e.clientHeight, scroll: e.scrollHeight }));
  await loc.screenshot({ path: `${OUT}/${file}` });
  log(`   screenshot: ${file} (${sel} on ${vp.width}x3600; clientHeight=${m.client} scrollHeight=${m.scroll})`);
  await page.setViewportSize(vp);
  await page.waitForTimeout(200);
}
const rule = (page: Page, id: string) => page.locator(`#network-slot .net-rule[data-rule="${id}"]`);
const headings = (page: Page) =>
  page.evaluate(() =>
    [...document.querySelectorAll<HTMLElement>("#network-slot .net-rule")].map(
      (s) => s.dataset.rule + ": " + s.querySelector("h3")!.textContent!.replace(/\s+/g, " ").trim() + "  [aria-text-content=" + JSON.stringify(s.querySelector("h3")!.textContent) + "]",
    ),
  );
async function openMap(page: Page) {
  await page.goto("/map.html");
  await mapReady(page);
  await expect(page.locator("#network-slot .net-rule")).toHaveCount(5);
}
async function finding(page: Page, id: string, file: string, log: (s: string) => void, pick?: RegExp) {
  const li = rule(page, id).locator("li[data-severity]", pick ? { hasText: pick } : undefined).first();
  await li.scrollIntoViewIfNeeded();
  await li.screenshot({ path: `${OUT}/${file}` });
  const info = await li.evaluate((e) => ({
    severity: e.getAttribute("data-severity"),
    why: e.querySelector("b")!.textContent,
    word: e.querySelector(".net-severity")!.textContent,
    links: [...e.querySelectorAll("a")].map((a) => `${a.textContent}=>${a.getAttribute("href")}`),
    more: e.querySelector(".net-more-places")?.textContent ?? null,
  }));
  log(`   ${id} finding: severity=${info.severity} (${info.word}) why="${info.why}" links=${info.links.length} ${JSON.stringify(info.links)} more=${info.more}`);
  log(`   screenshot: ${file}`);
}
const summary = (page: Page, log: (s: string) => void, tag: string) =>
  page.evaluate(() => ({
    lead: document.querySelector("#network-lead")!.textContent,
    status: document.querySelector("#fit-slot")!.textContent,
    checks: document.querySelector("#checks-lead")!.textContent,
  })).then((s) => log(`   [${tag}] network-lead="${s.lead}" | status line (#fit-slot)="${norm(s.status!)}" | Notes lead (#checks-lead)="${norm(s.checks!)}"`));

test("edit removes a finding: step 6", async ({ page }) => {
  const log = mk("d");
  log("6. Delete all bike lanes of a street at the head of the smallest cut-off cycling link, reload the map");
  await openMap(page);
  const target = await page.evaluate(() => {
    const items = [...document.querySelectorAll<HTMLElement>('#network-slot .net-rule[data-rule="cycling"] ul.places-list > li')];
    const islands = items.filter((li) => /cut off/.test(li.querySelector("b")!.textContent!)).map((li) => ({ links: [...li.querySelectorAll("a")].map((a) => a.getAttribute("href")!), text: li.querySelector("b")!.textContent! }));
    islands.sort((a, b) => a.links.length - b.links.length);
    return { href: islands[0].links[0], all: islands[0].links, text: islands[0].text };
  });
  const before = norm(await rule(page, "cycling").innerText());
  const beforeHead = norm((await rule(page, "cycling").locator("h3").textContent())!);
  log(`   BEFORE: cycling heading="${beforeHead}"; target finding "${target.text}" links=${JSON.stringify(target.all)}; edited street=${target.href}`);
  log(`   BEFORE: link to ${target.href} present in cycling list: ${await rule(page, "cycling").locator(`a[href="${target.href}"]`).count()}`);
  await summary(page, log, "before");
  await full(page, "#network-slot .net-rule[data-rule=cycling]", "09-before-cycling.png", log);
  await page.goto("/" + target.href);
  const bikes = page.locator("#drawing [data-role='seg']:has(rect.k-bike)");
  log(`   street page: bike lanes before: ${await bikes.count()}`);
  await page.screenshot({ path: `${OUT}/09b-street-before.png` });
  log("   screenshot: 09b-street-before.png");
  for (let guard = 0; (await bikes.count()) > 0 && guard < 4; guard++) {
    const index = await page.locator("#drawing [data-role='seg']").evaluateAll((els) => els.findIndex((e) => e.querySelector("rect.k-bike")));
    await page.locator("#wrap").focus();
    await page.keyboard.press("Escape");
    for (let i = 0; i <= index; i++) await page.keyboard.press("ArrowRight");
    await page.keyboard.press("Delete");
  }
  await expect(bikes).toHaveCount(0);
  log(`   street page: bike lanes after: ${await bikes.count()}`);
  await page.screenshot({ path: `${OUT}/09c-street-after.png` });
  log("   screenshot: 09c-street-after.png");
  await page.goto("/map.html");
  await mapReady(page);
  await expect(page.locator("#network-slot .net-rule")).toHaveCount(5);
  const after = norm(await rule(page, "cycling").innerText());
  log(`   AFTER: cycling heading="${norm((await rule(page, "cycling").locator("h3").textContent())!)}"; link to ${target.href} present: ${await rule(page, "cycling").locator(`a[href="${target.href}"]`).count()}; text changed: ${after !== before}`);
  log(`   AFTER lead: "${await page.locator("#network-lead").textContent()}"`);
  await summary(page, log, "after");
  log(`   AFTER findings texts: ${JSON.stringify(await rule(page, "cycling").locator("li[data-severity] b").allInnerTexts())}`);
  await full(page, "#network-slot .net-rule[data-rule=cycling]", "10-after-cycling.png", log);
  await full(page, "#network-panel", "10b-after-panel.png", log);
});
